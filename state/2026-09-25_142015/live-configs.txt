===== /home/jglerner/.config/wireplumber/wireplumber.conf.d/51-output-priorities.conf
# USB DACs: force analog profile, and stop WirePlumber from auto-switching to iec958
monitor.alsa.rules = [
  { matches = [ { device.name = "alsa_card.usb-FIIO_FiiO_K11_R2R-01" }
                { device.name = "~alsa_card.usb-HIFIMAN-EF499.*" } ]
    actions = { update-props = {
        device.profile       = "output:analog-stereo"
        api.acp.auto-profile = false
        api.acp.auto-port    = false } } }

  # Higher priority = preferred default output
  { matches = [ { node.name = "~alsa_output.usb-FIIO_FiiO_K11_R2R.*" } ]
    actions = { update-props = { priority.session = 2000 node.description = "FiiO K11 R2R (PO: ATH-R50x / LO: TOMATE)" } } }
  { matches = [ { node.name = "~alsa_output.usb-HIFIMAN-EF499.*" } ]
    actions = { update-props = { priority.session = 1900 node.description = "HIFIMAN EF499 (HD650)" } } }
  { matches = [ { node.name = "~alsa_output.pci-0000_65_00.1.*" } ]
    actions = { update-props = { priority.session = 100 node.description = "Monitor HDMI" } } }
]
